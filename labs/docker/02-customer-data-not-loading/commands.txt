# Both containers are up and the database is healthy, which is the fact that
# makes this interesting rather than the fact that solves it.
#
# The app's own health probe reads the database, so it fails here and the
# container settles on `unhealthy` after ten attempts three seconds apart. A
# fixed twelve second wait recorded `health: starting`, which is the phase
# before any of that has been decided, and the check would report drift the
# first time a runner got there a little later. Measured: still starting at 30
# seconds, unhealthy by 40.
#
# `keep: healthy` did not catch it, because `unhealthy` contains `healthy`.
# Naming the settled word is the marker actually doing its job.
# alias: docker ps -a
# alias: docker ps
# alias: docker compose ps
# wait: 12
# until: (unhealthy)
# keep: unhealthy, proveit-docker-app-1
docker ps -a --filter "label=com.docker.compose.project=proveit-docker"
---
# The symptom the customer reported, reproduced.
# alias: curl -i http://127.0.0.1:8100/customers
# alias: curl http://127.0.0.1:8100/customers
# keep: 503
curl -s -i http://127.0.0.1:8100/customers | head -12
---
# Read from the top of the log rather than the bottom of it.
#
# The health probe runs every three seconds forever, so `--tail 15` is a window
# sliding over a growing file and what it holds depends on the moment it is
# asked. That was hidden while the recording was made twelve seconds in, when
# the whole log was still shorter than fifteen lines. Waiting for the settled
# state exposed it: the same command then returned fifteen copies of the same
# pair and no sign of the process starting.
#
# The first lines of a process log do not move. This is also the better story:
# the server comes up, reaches for the database, and fails on the first try.
# alias: docker compose logs app
# keep: database_connection_failed, server_started
docker compose -f labs/docker/_stack/compose.yaml \
               -f labs/docker/_stack/compose.override.yaml \
               logs app | head -9
---
# What the application is aiming at, asked of the running process rather than
# read out of a file.
# alias: docker compose exec app printenv DB_HOST
# keep: localhost
docker compose -f labs/docker/_stack/compose.yaml \
               -f labs/docker/_stack/compose.override.yaml \
               exec -T app printenv DB_HOST
---
# What that name resolves to from inside the container, which is the only
# place it means anything.
# alias: docker compose exec app getent hosts localhost
# keep: localhost
docker compose -f labs/docker/_stack/compose.yaml \
               -f labs/docker/_stack/compose.override.yaml \
               exec -T app getent hosts localhost
---
# And the name Compose actually publishes for the database.
# alias: docker compose exec app getent hosts postgres
# keep: postgres
docker compose -f labs/docker/_stack/compose.yaml \
               -f labs/docker/_stack/compose.override.yaml \
               exec -T app getent hosts postgres
---
# The database was never down. Worth proving rather than assuming.
# keep: healthy
docker inspect --format '{{.State.Health.Status}}' proveit-docker-postgres-1
