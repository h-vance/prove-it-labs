# What the joins actually produce before anything is counted. The same customer
# appears once per user.
# keep: Customer 1
docker compose -f labs/sql/_stack/compose.yaml exec -T \
  -e PGPASSWORD=demo-password postgres \
  psql -U support -d support_lab -A -F'|' --pset=footer=off \
  -c "SELECT c.id, c.name, u.email FROM customers c JOIN workspaces w ON w.customer_id = c.id JOIN users u ON u.workspace_id = w.id ORDER BY c.id LIMIT 12;"

# The inflation factor is not random, and that is what identifies it: each plan
# is multiplied by how many users its accounts have.
# keep: enterprise
docker compose -f labs/sql/_stack/compose.yaml exec -T \
  -e PGPASSWORD=demo-password postgres \
  psql -U support -d support_lab -A -F'|' --pset=footer=off \
  -c "SELECT c.plan, count(DISTINCT c.id) AS customers, count(*) AS result_rows FROM customers c JOIN workspaces w ON w.customer_id = c.id JOIN users u ON u.workspace_id = w.id GROUP BY c.plan ORDER BY c.plan;"
---
