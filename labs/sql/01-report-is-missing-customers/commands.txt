# Which accounts the report never sees. Six of them, and the join is the only
# thing standing between them and the count.
# keep: 6
docker compose -f labs/sql/_stack/compose.yaml exec -T \
  -e PGPASSWORD=demo-password postgres \
  psql -U support -d support_lab -A -F'|' --pset=footer=off \
  -c "SELECT c.id, c.name, c.plan FROM customers c LEFT JOIN workspaces w ON w.customer_id = c.id WHERE w.id IS NULL ORDER BY c.id;"
---
# The pattern the finance team noticed, confirmed. Uneven loss has a mechanism
# behind it; even loss usually does not.
# keep: starter
docker compose -f labs/sql/_stack/compose.yaml exec -T \
  -e PGPASSWORD=demo-password postgres \
  psql -U support -d support_lab -A -F'|' --pset=footer=off \
  -c "SELECT c.plan, count(*) AS without_workspace FROM customers c LEFT JOIN workspaces w ON w.customer_id = c.id WHERE w.id IS NULL GROUP BY c.plan ORDER BY c.plan;"
---
