# 300,000 rows read to return 7,500. The cost of that is the size of the table,
# and the table grows every day.
# keep: 300000
docker compose -f labs/sql/_stack/compose.yaml exec -T \
  -e PGPASSWORD=demo-password postgres \
  psql -U support -d support_lab -A --pset=footer=off \
  -c "SELECT count(*) AS total_rows FROM api_requests;"
