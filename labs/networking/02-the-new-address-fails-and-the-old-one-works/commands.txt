# The nightly upload on the address the customer was migrated to.
#
# The exit code and the closing paragraph are the same ones the expiry produced
# in the previous exercise. The line above them is not.
# alias: /app/upload.sh
# alias: docker compose exec client /app/upload.sh
# wait: 10
# keep: no alternative certificate subject name, reports, upload failed
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client /app/upload.sh
---
# The same client, the same credentials, the same payload, the other address.
# One gateway is behind both of them.
# alias: GATEWAY_URL=https://gateway:8443 /app/upload.sh
# alias: upload.sh against gateway
# keep: accepted, upload finished
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client sh -c 'GATEWAY_URL=https://gateway:8443 /app/upload.sh'
---
# Both names resolve to the same address, which is what rules out a second
# server before anybody goes looking for one.
# alias: getent hosts reports gateway
# alias: getent hosts reports
# keep: reports, gateway
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client sh -c 'getent hosts reports; getent hosts gateway'
---
# What is being served on the new address. The dates are exactly what the
# customer said they were. The last line is the one that matters.
# alias: openssl x509 -ext subjectAltName
# alias: openssl s_client -connect reports:8443
# keep: notAfter=Dec, DNS:gateway
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client sh -c 'echo | openssl s_client -connect reports:8443 -servername reports 2>/dev/null | openssl x509 -noout -subject -dates -ext subjectAltName'
---
# Every version the CA has issued, read for names rather than for dates.
#
# v1 expired in April and covered both names. v2 replaced it and covers one.
# The rotation that fixed the previous incident is what opened this one.
# alias: ls /certs
# alias: openssl x509 -in /certs/v1.pem -noout -ext subjectAltName
# keep: v1.pem, v2.pem, v3.pem, DNS:gateway, DNS:reports
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T gateway sh -c 'for c in /certs/v*.pem; do echo "$c"; openssl x509 -in "$c" -noout -dates -ext subjectAltName; done'
