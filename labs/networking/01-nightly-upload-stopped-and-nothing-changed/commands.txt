# What the gateway logged while two uploads were failing.
#
# One line, from startup. This reads like a dead end and is the opposite: a
# request that had arrived would be here, so no request was ever sent. The
# caller hung up before asking for anything.
# alias: docker compose logs gateway
# alias: logs gateway
# keep: server_started
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               logs gateway
---
# So ask the port what it presents, before any request is sent.
# alias: openssl s_client | openssl x509
# alias: openssl x509 -noout -dates
# keep: notBefore=Jan, notAfter=Apr, CN=gateway
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client sh -c 'echo | openssl s_client -connect gateway:8443 -servername gateway 2>/dev/null | openssl x509 -noout -subject -dates -ext subjectAltName'
---
# The connection on its own, asked to treat a bad chain as fatal. The
# verification failure is stated twice, once as the chain is walked and once in
# the summary, and `-verify_return_error` is what stops after the second one.
#
# The flag is not decoration. Plain `s_client` completes the handshake and then
# keeps the socket open, and everything it prints after that point moves:
#
#   - TLS 1.3 delivers the session ticket after the handshake, so whether it
#     arrives before `echo` closes the connection is a race. Each one that
#     lands prints another 34 lines. Fifteen runs returned three lengths:
#     63, 97 and 131.
#   - The chain walk goes to standard error and the summary to standard output.
#     Merging them puts the two blocks in whichever order they flush, which is
#     not fixed either. That is what CI caught: one run in eight swapped them.
#
# Failing on the bad chain ends the connection before any of that, so every run
# is identical. What is lost is the body of the certificate, which is the one
# part of this output that is not evidence.
# alias: openssl s_client -verify_return_error -connect gateway:8443
# alias: openssl s_client -verify_return_error
# keep: verify error:num=10, Verify return code: 10, NotAfter: Apr
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T client sh -c 'echo | openssl s_client -verify_return_error -connect gateway:8443 -servername gateway'
---
# Every version the internal CA has issued, with the dates on each.
#
# Worth doing rather than reaching for the highest number. Reading the dates is
# the habit; assuming the newest file is the right one is how the next exercise
# starts.
# alias: ls /certs
# alias: openssl x509 -in /certs/v2.pem -noout -dates
# keep: v1.pem, v2.pem, v3.pem
docker compose -f labs/networking/_stack/compose.yaml \
               -f labs/networking/_stack/compose.override.yaml \
               exec -T gateway sh -c 'for c in /certs/v*.pem; do echo "$c"; openssl x509 -in "$c" -noout -dates -ext subjectAltName; done'
