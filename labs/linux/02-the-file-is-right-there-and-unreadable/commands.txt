# alias: docker compose exec worker python3 /app/reporter.py
# wait: 15
# keep: Permission denied, running_as_uid, supplementary_groups
docker compose -f labs/linux/_stack/compose.yaml \
               -f labs/linux/_stack/compose.override.yaml \
               exec -T worker python3 /app/reporter.py

# Owner, group and mode without the noise, since those three are the question.
# alias: stat /etc/reporting/credentials.conf
# keep: owner=root, group=reporting, mode=640
docker compose -f labs/linux/_stack/compose.yaml \
               -f labs/linux/_stack/compose.override.yaml \
               exec -T worker stat -c '%n owner=%U group=%G mode=%a' /etc/reporting/credentials.conf
---
